local function CreateMatrixEffect(parentFrame, columnCount, speedMultiplier)
    local MatrixContainer = Instance.new("Frame")
    MatrixContainer.Name = "MatrixContainer"
    MatrixContainer.Size = UDim2.new(1, 0, 1, 0)
    MatrixContainer.BackgroundTransparency = 1
    MatrixContainer.ClipsDescendants = true
    MatrixContainer.ZIndex = parentFrame.ZIndex + 1
    MatrixContainer.Parent = parentFrame

    local running = true

    task.spawn(function()
        while running do
            local col = math.random(1, columnCount)
            local posX = (col - 1) / columnCount

            local MatrixText = Instance.new("TextLabel")
            MatrixText.Size = UDim2.new(1 / columnCount, 0, 0, 16)
            MatrixText.Position = UDim2.new(posX, 0, 0, -20)
            MatrixText.BackgroundTransparency = 1
            MatrixText.Text = tostring(math.random(0, 1))
            
            if math.random(1, 3) == 1 then
                MatrixText.TextColor3 = Color3.fromRGB(240, 240, 245)
            else
                MatrixText.TextColor3 = Color3.fromRGB(140, 140, 150)
            end
            
            MatrixText.TextTransparency = math.random(1, 4) / 10
            MatrixText.TextSize = math.random(11, 15)
            MatrixText.Font = Enum.Font.Code
            MatrixText.ZIndex = MatrixContainer.ZIndex
            MatrixText.Parent = MatrixContainer

            local fallSpeed = (math.random(8, 18) / 10) * (speedMultiplier or 1)
            local tween = TweenService:Create(MatrixText, TweenInfo.new(fallSpeed, Enum.EasingStyle.Linear), {
                Position = UDim2.new(posX, 0, 1, 20),
                TextTransparency = 1
            })
            
            tween:Play()
            tween.Completed:Connect(function()
                MatrixText:Destroy()
            end)

            task.wait(0.005)
            if not parentFrame.Parent then break end
        end
    end)

    return function()
        running = false
        MatrixContainer:Destroy()
    end
end

local function EnableSmoothDrag(guiObject, dragHandle)
    dragHandle = dragHandle or guiObject
    local dragging = false
    local dragStart = Vector3.new()
    local startPos = UDim2.new()
    local hasMoved = false

    dragHandle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            hasMoved = false
            dragStart = input.Position
            startPos = guiObject.Position
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            if delta.Magnitude > 6 then
                hasMoved = true
            end
            
            guiObject.Position = UDim2.new(
                startPos.X.Scale, 
                startPos.X.Offset + delta.X, 
                startPos.Y.Scale, 
                startPos.Y.Offset + delta.Y
            )
        end
    end)

    local function StopDrag()
        dragging = false
    end

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            StopDrag()
        end
    end)

    return function() return hasMoved end
end

local WasIconDragged = EnableSmoothDrag(MiniIcon, MiniIcon)

local isAnimating = false
local rotationConnection = nil

local function StartIconRotation()
    if rotationConnection then rotationConnection:Disconnect() end
    rotationConnection = RunService.RenderStepped:Connect(function(deltaTime)
        MiniIcon.Rotation = (MiniIcon.Rotation + (120 * deltaTime)) % 360
    end)
end

local function StopIconRotation()
    if rotationConnection then
        rotationConnection:Disconnect()
        rotationConnection = nil
    end
end

local function MinimizeUI()
    if isAnimating then return end
    isAnimating = true

    Container.Visible = false
    TopBar.Visible = false

    local shrinkTween = TweenService:Create(MainFrame, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.In), {
        Size = UDim2.new(0, 0, 0, 0),
        BackgroundTransparency = 1
    })
    shrinkTween:Play()

    shrinkTween.Completed:Connect(function()
        MainFrame.Visible = false
        MiniIcon.Visible = true
        MiniIcon.Size = UDim2.new(0, 0, 0, 0)
        
        StartIconRotation()
        
        local popTween = TweenService:Create(MiniIcon, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Size = UDim2.new(0, 48, 0, 48)
        })
        popTween:Play()
        popTween.Completed:Connect(function()
            isAnimating = false
        end)
    end)
end

local function ExpandUI()
    if WasIconDragged() or isAnimating then return end
    isAnimating = true

    local iconShrink = TweenService:Create(MiniIcon, TweenInfo.new(0.25, Enum.EasingStyle.Back, Enum.EasingDirection.In), {
        Size = UDim2.new(0, 0, 0, 0)
    })
    iconShrink:Play()

    iconShrink.Completed:Connect(function()
        StopIconRotation()
        MiniIcon.Visible = false
        
        MainFrame.Visible = true
        local expandTween = TweenService:Create(MainFrame, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Size = UDim2.new(0, 380, 0, 240),
            BackgroundTransparency = 0
        })
        expandTween:Play()

        expandTween.Completed:Connect(function()
            Container.Visible = true
            TopBar.Visible = true
            isAnimating = false
        end)
    end)
end
